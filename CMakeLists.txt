cmake_minimum_required(VERSION 3.22)
project(PiradexMV VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# JUCE via FetchContent (descarregado na 1ª build — não precisas de o ter instalado)
include(FetchContent)
FetchContent_Declare(
    JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG        8.0.4
    GIT_SHALLOW    TRUE
)
FetchContent_MakeAvailable(JUCE)

juce_add_plugin(PiradexMV
    COMPANY_NAME             "BeatFreak"
    PLUGIN_MANUFACTURER_CODE  Bfrk
    PLUGIN_CODE               Pmv2
    FORMATS                   AU VST3 Standalone
    PRODUCT_NAME             "LEVEL-X"
    BUNDLE_ID                "com.beatfreak.levelx"
    IS_SYNTH                  FALSE
    NEEDS_MIDI_INPUT          FALSE
    NEEDS_MIDI_OUTPUT         FALSE
    COPY_PLUGIN_AFTER_BUILD   TRUE
)

juce_generate_juce_header(PiradexMV)

target_sources(PiradexMV PRIVATE
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp
)

target_compile_definitions(PiradexMV PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_DISPLAY_SPLASH_SCREEN=0
)

target_link_libraries(PiradexMV
    PRIVATE
        juce::juce_audio_utils
        juce::juce_dsp
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags
)
